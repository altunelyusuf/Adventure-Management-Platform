### SPARQL Queries
```sparql
# Find all active quests near a location
PREFIX ac-quest: <http://example.org/adventure-coordinator/quest#>
PREFIX ac-geo: <http://example.org/adventure-coordinator/geospatial#>

SELECT ?quest ?title ?distance
WHERE {
  ?quest a ac-quest:Quest ;
         ac:hasState ac-quest:Active ;
         ac-quest:questTitle ?title ;
         ac-geo:hasLocation ?location .
  
  ?location ac-geo:latitude ?lat ;
            ac-geo:longitude ?long .
  
  BIND(distance(?lat, ?long, 40.7128, -74.0060) AS ?distance)
  FILTER(?distance < 5.0)
}
ORDER BY ?distance
LIMIT 20
```
